```sql
CREATE TABLE `sys_user` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `username` varchar(50) NOT NULL COMMENT '用户名',
  `password` varchar(100) NOT NULL COMMENT 'BCrypt 加密密码',
  `nickname` varchar(50) DEFAULT NULL COMMENT '昵称',

  `sort` int DEFAULT '0',
  `status` tinyint DEFAULT '1',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_role_key` (`role_key`)

  `id` bigint NOT NULL AUTO_INCREMENT,
  `parent_id` bigint DEFAULT '0',
  `name` varchar(50) NOT NULL,
  `path` varchar(200) DEFAULT NULL,
  `component` varchar(255) DEFAULT NULL,
  `permission` varchar(100) DEFAULT NULL COMMENT '权限标识 sys:user:add',
  `type` tinyint NOT NULL DEFAULT '0' COMMENT '0 目录 1 菜单 2 按钮',
  `icon` varchar(50) DEFAULT NULL,
  `sort` int DEFAULT '0',
  `visible` tinyint DEFAULT '1',
  PRIMARY KEY (`id`)
);

-- 4. 用户角色关联表 (多对多)
CREATE TABLE `sys_user_role` (
  `user_id` bigint NOT NULL,
  `role_id` bigint NOT NULL,
  PRIMARY KEY (`user_id`,`role_id`)
);

-- 5. 角色菜单关联表 (多对多)
CREATE TABLE `sys_role_menu` (
  `role_id` bigint NOT NULL,
  `menu_id` bigint NOT NULL,
  PRIMARY KEY (`role_id`,`menu_id`)
);
```

#### 2.1.2 设计思路

**RBAC 模型（Role-Based Access Control）**

```
用户 (User) ←→ 角色 (Role) ←→ 菜单/权限 (Menu/Permission)
```

- 一个用户可以拥有多个角色
- 一个角色可以拥有多个菜单/权限
- 权限校验时：用户 → 角色 → 权限集合

## 三、Spring Security + JWT 认证配置

### 3.1 核心配置类 SecurityConfig

```java
@Configuration
@EnableWebSecurity
@EnableMethodSecurity(prePostEnabled = true)  // 开启方法级权限控制
public class SecurityConfig {

    @Autowired
    private SysUserService sysUserService;

    @Autowired
    private JwtAuthenticationFilter jwtAuthenticationFilter;

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
            // 1. 禁用 CSRF（JWT 不需要）
            .csrf().disable()
            
            // 2. 禁用 Session（无状态认证）
            .sessionManagement()
            .sessionCreationPolicy(SessionCreationPolicy.STATELESS)
            
            .and()
            // 3. 配置 CORS（允许跨域）
            .cors()
            
            .and()
            // 4. 授权配置
            .authorizeHttpRequests()
            // 放行登录接口
            .requestMatchers("/api/system/login", "/api/system/logout").permitAll()
            // 放行 Swagger 文档
            .requestMatchers("/doc.html", "/webjars/**").permitAll()
            // 放行 OPTIONS 请求
            .requestMatchers(HttpMethod.OPTIONS, "/**").permitAll()
            // 其他请求需要认证
            .anyRequest().authenticated();

        // 5. 添加 JWT 过滤器（在用户名密码过滤器之前）
        http.addFilterBefore(jwtAuthenticationFilter, 
                            UsernamePasswordAuthenticationFilter.class);

        return http.build();
    }

    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder(10);  // salt rounds = 10
    }

    @Bean
    public AuthenticationProvider authenticationProvider() {
        DaoAuthenticationProvider provider = new DaoAuthenticationProvider();
        provider.setUserDetailsService(sysUserService);
        provider.setPasswordEncoder(passwordEncoder());
        return provider;
    }
}
```

### 3.2 设计要点

| 配置项 | 作用 | 为什么这样配置 |
|--------|------|----------------|
| `csrf().disable()` | 禁用 CSRF 防护 | JWT 是无状态的，不依赖 Cookie，不存在 CSRF 风险 |
| `STATELESS` | 不创建 Session | 减少服务器内存消耗，支持水平扩展 |
| `cors()` | 启用跨域 | 前后端分离，前端运行在不同域名/端口 |
| `permitAll()` | 放行特定接口 | 登录接口需要公开访问 |
| `addFilterBefore` | 添加 JWT 过滤器 | 在 Spring Security 处理前解析 Token |

## 四、JWT 工具类实现

### 4.1 JWT 工具类

```java
@Component
public class JwtUtils {

    @Value("${jwt.secret}")
    private String secret;

    @Value("${jwt.expiration}")
    private Long expiration;  // 24 小时 = 86400000ms

    /**
     * 生成 JWT Token
     */
    public String generateToken(Long userId, String username) {
        Map<String, Object> claims = new HashMap<>();
        claims.put("userId", userId);
        claims.put("username", username);
        return createToken(claims, username);
    }

    private String createToken(Map<String, Object> claims, String subject) {
        Date now = new Date();
        Date expirationDate = new Date(now.getTime() + expiration);
        return Jwts.builder()
                .setClaims(claims)
                .setSubject(subject)
                .setIssuedAt(now)
                .setExpiration(expirationDate)
                .signWith(SignatureAlgorithm.HS512, secret)
                .compact();
    }

    /**
     * 从 Token 获取用户 ID
     */
    public Long getUserIdFromToken(String token) {
        Claims claims = getClaimsFromToken(token);
        return Long.valueOf(claims.get("userId").toString());
    }

    /**
     * 验证 Token 有效性
     */
    public boolean validateToken(String token) {
        try {
            Jwts.parser().setSigningKey(secret).parseClaimsJws(token);
            return !isTokenExpired(token);
        } catch (Exception e) {
            return false;
        }
    }

    private Claims getClaimsFromToken(String token) {
        return Jwts.parser()
                .setSigningKey(secret)
                .parseClaimsJws(token)
                .getBody();
    }
}
```

### 4.2 JWT 结构

```
Header: {"alg": "HS512", "typ": "JWT"}
Payload: {
    "userId": 1,
    "username": "admin",
    "iat": 1713158400,
    "exp": 1713244800
}
Signature: HMACSHA512(base64(header) + "." + base64(payload), secret)
```

## 五、JWT 认证过滤器

### 5.1 过滤器实现

```java
@Component
public class JwtAuthenticationFilter extends OncePerRequestFilter {

    @Resource
    private JwtUtils jwtUtils;

    @Resource
    private SysUserService sysUserService;

    @Override
    protected void doFilterInternal(HttpServletRequest request, 
                                    HttpServletResponse response,
                                    FilterChain filterChain) throws ServletException, IOException {
        try {
            // 1. 从请求头获取 Token
            String token = getTokenFromRequest(request);

            // 2. 验证 Token
            if (StringUtils.hasText(token) && jwtUtils.validateToken(token)) {
                // 3. 从 Token 中获取用户 ID
                Long userId = jwtUtils.getUserIdFromToken(token);

                // 4. 加载用户信息
                SysUser user = sysUserService.getById(userId);
                
                // 5. 检查用户状态
                if (user != null && user.getStatus() == 1) {
                    // 6. 设置认证信息到 SecurityContext
                    UsernamePasswordAuthenticationToken authentication =
                        new UsernamePasswordAuthenticationToken(
                            user, null, new ArrayList<>());
                    authentication.setDetails(
                        new WebAuthenticationDetailsSource().buildDetails(request));
                    SecurityContextHolder.getContext().setAuthentication(authentication);
                }
            }
        } catch (Exception e) {
            log.error("JWT 认证失败：{}", e.getMessage());
        }

        // 7. 继续执行过滤器链
        filterChain.doFilter(request, response);
    }

    private String getTokenFromRequest(HttpServletRequest request) {
        String bearerToken = request.getHeader("Authorization");
        if (StringUtils.hasText(bearerToken) && bearerToken.startsWith("Bearer ")) {
            return bearerToken.substring(7);
        }
        return null;
    }
}
```

### 5.2 过滤器执行流程

```
请求 → JwtAuthenticationFilter → SecurityFilterChain → Controller
         ↓
    解析 Token → 验证 → 设置 Authentication → SecurityContext
```

**关键点：**
- `OncePerRequestFilter`：确保每次请求只执行一次
- `SecurityContextHolder`：ThreadLocal 存储认证信息
- `filterChain.doFilter()`：必须调用，否则请求链中断

### 6.2 Service 层（核心业务逻辑）

```java
@Slf4j
@Service
public class SysUserServiceImpl extends ServiceImpl<SysUserMapper, SysUser> 
                                 implements SysUserService {

    @Autowired
    private PasswordEncoder passwordEncoder;

    @Autowired
    private JwtUtils jwtUtils;

    @Autowired
    private RedisUtils redisUtils;

    @Autowired
    private SysMenuService sysMenuService;

    @Override
    public LoginResponse login(LoginRequest request) {
        // 1. 查询用户
        SysUser user = this.getOne(new LambdaQueryWrapper<SysUser>()
                .eq(SysUser::getUsername, request.getUsername()));

        if (user == null) {
            throw new BusinessException(ResultCode.USER_NOT_FOUND);
        }

        // 2. 验证密码（BCrypt 匹配）
        if (!passwordEncoder.matches(request.getPassword(), user.getPassword())) {
            throw new BusinessException(ResultCode.USER_PASSWORD_ERROR);
        }

        // 3. 检查用户状态
        if (user.getStatus() == 0) {
            throw new BusinessException(ResultCode.USER_DISABLED);
        }

        // 4. 生成 JWT Token
        String token = jwtUtils.generateToken(user.getId(), user.getUsername());

        // 5. Token 存入 Redis（支持主动失效）
        String redisKey = "token:" + user.getId();
        redisUtils.set(redisKey, token, 24 * 60 * 60);

        // 6. 获取权限和菜单
        List<String> permissions = getUserPermissions(user.getId());
        List<MenuVO> menus = sysMenuService.getTreeListByUserId(user.getId());

        // 7. 构建响应
        LoginResponse response = new LoginResponse();
        response.setToken(token);
        response.setUsername(user.getUsername());
        response.setPermissions(permissions);
        response.setMenus(menus);

        return response;
    }

    @Override
    public void logout(String token) {
        Long userId = jwtUtils.getUserIdFromToken(token);
        redisUtils.del("token:" + userId);  // 使 Token 失效
    }

    @Override
    public SysUser getCurrentUser() {
        Authentication auth = SecurityContextHolder.getContext().getAuthentication();
        return (auth.getPrincipal() instanceof SysUser) 
                ? (SysUser) auth.getPrincipal() : null;
    }
}
```

### 7.2 全局异常处理器

```java
@RestControllerAdvice
@Slf4j
public class GlobalExceptionHandler {

    /**
     * 业务异常
     */
    @ExceptionHandler(BusinessException.class)
    public Result<Void> handleBusinessException(BusinessException e) {
        log.warn("业务异常：{}", e.getMessage());
        return Result.error(e.getCode(), e.getMessage());
    }

    /**
     * 参数校验异常
     */
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public Result<Void> handleValidationException(
            MethodArgumentNotValidException e) {
        String message = e.getBindingResult().getFieldErrors().stream()
                .map(error -> error.getField() + ": " + error.getDefaultMessage())
                .collect(Collectors.joining(", "));
        return Result.error(400, "参数错误：" + message);
    }

    /**
     * 其他异常
     */
    @ExceptionHandler(Exception.class)
    public Result<Void> handleException(Exception e) {
        log.error("系统异常", e);
        return Result.error("系统异常：" + e.getMessage());
    }
}
```

### 8.3 动态菜单加载

```java
@Service
public class SysMenuServiceImpl extends ServiceImpl<SysMenuMapper, SysMenu> 
                                 implements SysMenuService {

    /**
     * 根据用户 ID 获取菜单树
     */
    @Override
    public List<MenuVO> getTreeListByUserId(Long userId) {
        // 1. 获取用户所有菜单（平铺）
        List<SysMenu> allMenus = baseMapper.selectMenusByUserId(userId);

        // 2. 转换为树形结构
        return buildMenuTree(allMenus, 0L);
    }

    private List<MenuVO> buildMenuTree(List<SysMenu> menus, Long parentId) {
        return menus.stream()
                .filter(m -> m.getParentId().equals(parentId))
                .map(menu -> {
                    MenuVO vo = new MenuVO();
                    BeanUtils.copyProperties(menu, vo);
                    // 递归加载子菜单
                    vo.setChildren(buildMenuTree(menus, menu.getId()));
                    return vo;
                })
                .collect(Collectors.toList());
    }
}
```

## 十、Redis 集成

### 10.1 Redis 配置类

```java
@Configuration
public class RedisConfig {

    @Bean
    public RedisTemplate<String, Object> redisTemplate(
            RedisConnectionFactory factory) {
        RedisTemplate<String, Object> template = new RedisTemplate<>();
        template.setConnectionFactory(factory);

        // JSON 序列化配置
        ObjectMapper mapper = new ObjectMapper();
        mapper.setVisibility(PropertyAccessor.ALL, 
                            JsonAutoDetect.Visibility.ANY);
        mapper.activateDefaultTyping(LaissezFaireSubTypeValidator.instance, 
                                    ObjectMapper.DefaultTyping.NON_FINAL);

        Jackson2JsonRedisSerializer<Object> jsonSerializer = 
                new Jackson2JsonRedisSerializer<>(Object.class);
        jsonSerializer.setObjectMapper(mapper);
        StringRedisSerializer stringSerializer = new StringRedisSerializer();

        // key 采用 String 序列化
        template.setKeySerializer(stringSerializer);
        template.setHashKeySerializer(stringSerializer);
        // value 采用 JSON 序列化
        template.setValueSerializer(jsonSerializer);
        template.setHashValueSerializer(jsonSerializer);

        template.afterPropertiesSet();
        return template;
    }
}
```

```java
     * 设置缓存（带过期时间）

     * 获取缓存

    public Object get(String key) {

    public void del(String key) {

    }

    /**

    public Boolean hasKey(String key) {
```

## 十三、面试常见问题

### Q1: 为什么选择 JWT 而不是 Session？

**答：**
1. **无状态**：JWT 不依赖服务器 Session，支持水平扩展
2. **跨域友好**：适合前后端分离架构
3. **性能更好**：不需要查询 Session 存储（Redis）
4. **自包含**：Token 中包含用户信息，减少数据库查询

**缺点：**
- Token 无法主动失效（本方案通过 Redis 存储解决）
- Token 较大，增加传输开销

### Q2: 如何防止 JWT 被篡改？

**答：**
1. **签名验证**：JWT 使用 HMACSHA512 签名，篡改后验证失败
2. **HTTPS 传输**：防止中间人攻击
3. **密钥保管**：secret 存储在服务器，不泄露

### Q3: RBAC 模型的优势？

**答：**
1. **灵活授权**：用户通过角色间接获得权限，便于管理
2. **最小权限原则**：可以精确控制每个角色的权限范围
3. **易于维护**：权限变更只需修改角色，不需要逐个修改用户

### Q4: BCrypt 加密的原理？

**答：**
1. **加盐**：每次加密自动生成随机 salt，防止彩虹表攻击
2. **自适应**：可以通过 rounds 参数调整计算复杂度
3. **不可逆**：只能验证匹配，不能解密

```java
// 加密
String encoded = passwordEncoder.encode("123456");
// 验证
boolean matches = passwordEncoder.matches("123456", encoded);
```

### Q5: 如何实现按钮级权限控制？

**答：**
**后端：**
```java
@PreAuthorize("hasAuthority('sys:user:add')")
public Result<Void> add(@RequestBody SysUser user) { }
```

**前端：**
```vue
<button v-permission="'sys:user:add'">新增</button>
```

**原理：**
1. 登录后返回权限列表 `['sys:user:add', 'sys:user:edit']`
2. 前端自定义指令检查权限是否存在
3. 后端拦截器验证权限标识
